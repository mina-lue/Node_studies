QUESTION 12 - Event order, once, and error (6 points)

Assume no process-level exception handler is installed.

var EventEmitter = require('events').EventEmitter;
var bus = new EventEmitter();
var total = 2;

function add(value) {
    total += value;
    console.log('add ' + total);
}

bus.on('step', add);
bus.once('step', function (value) {
    console.log('once ' + value + ' ' + total);
});

console.log('start');
bus.emit('step', 3);
bus.removeListener('step', add);
bus.emit('step', 4);
console.log('finish ' + total);
bus.emit('error', new Error('broken'));
console.log('after error');

a. Give the exact console.log output up to the error, in order. (3 points)
b. Explain listener order, why the second step emission behaves as it does,
   and whether the listeners run synchronously or are deferred. (2 points)
c. Explain what the error emission does, whether 'after error' is printed,
   and how the process terminates. Do not reproduce the stack trace. (1 point)

QUESTION 13 - Constructor calls and prototype methods (6 points)

function Label(name) {
    this.name = name;
}
Label.prototype.describe = function () {
    return 'label:' + this.name;
};

function Parcel(name) {
    Label.call(this, name);

Parcel.prototype.describe = function () {

first.name = 'north';

console.log(first.describe());

console.log(first instanceof Parcel);
console.log(first instanceof Label);
console.log(first.constructor === Parcel);

QUESTION 14 - Mounted routes and error flow (7 points)

var express = require('express');
var app = express();
var router = express.Router();

app.use(function (req, res, next) {
    console.log('outer ' + req.url);
    next();
});

router.use(function (req, res, next) {
    console.log('inner ' + req.url);
    console.log('original ' + req.originalUrl);
    next();
});

router.get('/books/:id', function (req, res, next) {
    console.log('book ' + req.params.id + ' ' + req.query.brief);
    next(new Error('catalog unavailable'));

    console.log('router fallback');
    res.end('fallback');
});

app.use('/api', router);
app.use(function (err, req, res, next) {
    console.log('handled ' + err.message);
    res.statusCode = 503;
    res.end('try later');
});
app.listen(3000);

a. Give the exact console.log output for this request, in order. (3 points)
b. Give the response status and body. (1 point)
c. Explain the two URL forms, how the route obtains id and brief, and why

QUESTION 15 - Errors across an asynchronous boundary (6 points)

Assume a GET request reaches this core Node.js HTTP server. The file read
completes asynchronously with an Error whose code is ENOENT. No process-level
exception handler is installed, and no other requests are in flight.

var http = require('http');
var fs = require('fs');

http.createServer(function (req, res) {
    try {
        fs.readFile('absent.txt', function (err, data) {
            if (err) {
                throw err;
            }
            res.end(data);
        });
        console.log('read scheduled');
    } catch (err) {
        console.log('caught');
        res.statusCode = 500;
        res.end('failed');
    }
}).listen(3000);

a. State the console.log output, whether the catch block runs, whether this
   code completes an HTTP error response, and what happens to the server
   process. Explain the execution boundary responsible. (3 points)
b. Rewrite the request handler to send a completed plain-text 404 response
   for ENOENT, a completed plain-text 500 response for any other read error,
   and the file contents on success. Every branch must send at most one
   response. (3 points)
