QUESTION 1 - Closure model (8 points)

Explain what a closure retains when it refers to a variable declared in an
outer function. Does it retain a fixed copy of the value at the moment the
callback was scheduled? What happens if the variable changes before the
callback runs? Contrast this with two calls to the same factory.

ANSWER:
Closures retain access to the outer variable itself not fixed value of 
the variable, if there is change to the value of the variable before the 
callback runs, the callback gets the updated value of the variable. For 
each invocation of factory we have separate state and variables.

# bin/cli.js

#!/usr/bin/env node

const inspect = require('../lib/inspect');

// Process CLI arguments (ignoring node and script path)
const args = process.argv.slice(2);

inspect(args);

# lib/inspect.js

const fs = require('fs');
function inspectSettings(args) { 
let isVerbose = false;
let filePath = null;
let extraArgs = [];

for( let i =0; i < args.length; i++) {
	const arg = args[i];
	if(arg === "-v") {
	isVerbose = true;
	} else if (!filePath) {
	filePath = arg;
	} else {
	extraArgs.push(arg);
	}
}

if(!filePath) {
	console.error('Error: Required JSON file path missing');
	process.exitCode = 1;
	return;
}

if(extraArgs.length > 0) {
	console.error('Error: Unexpected extra arguments' + extraArgs.join(' '));
	process.exitCode = 1;
	return;
}


fs.readFile(filePath, 'utf8', function (err, data) {
	if (err) {
      		console.error('Filesystem Error: ' + err.message);
      		process.exitCode = 1;
      		return;
    	}
	try {
      const settings = JSON.parse(data);

      // Success Path
      if (isVerbose) {
        console.log('[VERBOSE] Successfully loaded file:', filePath);
        console.log('[VERBOSE] Parsed settings content:', settings);
      } else {
        console.log(settings);
      }
    } catch (parseErr) {
      // Error Path 4: JSON Syntax/Parsing error
      console.error('JSON Error: File does not contain valid JSON.');
      process.exitCode = 1;
    }
});

}
module.exports = inspectSettings;


e. 
# The parser iterates through every argument in process.argv.slice(2). It
flags -v wherever it appears and assigns the first non-flag string to filePath.
Consequently whether -v precedes or follows settings.json , both values are 
correctly captured before validating positional argument.

# fs.readFile offloads the IO operation to Node.js's underlying libuv thread
pool. The main v8 JavaScript thread continues running without waiting for the 
file to read. Once the operating system completes the file read, libuv places
the callback function onto the Event Loop's callback queue, which executes 
non-blockingly once the call stack clears.
